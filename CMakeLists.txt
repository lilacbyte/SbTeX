cmake_minimum_required(VERSION 3.20)
project(SbTeX VERSION 0.2.0 LANGUAGES CXX)
include(GNUInstallDirs)
find_package(PkgConfig REQUIRED)
pkg_check_modules(CAIRO REQUIRED IMPORTED_TARGET cairo>=1.16)
add_executable(sbtex main.cpp cmdParser.cpp format.cpp)
target_compile_features(sbtex PRIVATE cxx_std_20)
target_link_libraries(sbtex PRIVATE PkgConfig::CAIRO)
target_compile_definitions(sbtex PRIVATE
    SBTEX_FONT_DIR="${CMAKE_INSTALL_FULL_DATADIR}/sbtex/fonts"
    SBTEX_SOURCE_FONT_DIR="${CMAKE_CURRENT_SOURCE_DIR}/fonts")
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(sbtex PRIVATE -Wall -Wextra -Wpedantic)
endif()
install(TARGETS sbtex RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(DIRECTORY fonts/ DESTINATION ${CMAKE_INSTALL_DATADIR}/sbtex/fonts)
include(CTest)
if(BUILD_TESTING)
    find_package(Python3 COMPONENTS Interpreter)
    if(Python3_Interpreter_FOUND)
        add_test(NAME cli COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/cli.py
            $<TARGET_FILE:sbtex> ${CMAKE_CURRENT_SOURCE_DIR}/fonts)
    endif()
endif()
